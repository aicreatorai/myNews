```go
    Left  *Tree[T]
    Right *Tree[T]
}

// 泛型递归约束的实际应用：表达更复杂的类型关系
func (t *Tree[T]) Insert(value T) *Tree[T] {
    if t == nil {
        return &Tree[T]{Value: value}
    }
    if value.Less(t.Value) {
        t.Left = t.Left.Insert(value)
    } else {
        t.Right = t.Right.Insert(value)
    }
    return t
}
```

**虚拟线程最佳实践**：

```java
// Java 25+ 虚拟线程的推荐用法
public class VirtualThreadExample {
    public static void main(String[] args) throws Exception {
        // 方式1：使用Executors.newVirtualThreadPerTaskExecutor()
        try (var executor = Executors.newVirtualThreadPerTaskExecutor()) {
            // 处理10000个并发任务
            // 虚拟线程每个仅消耗约2KB，而平台线程需要1MB+
            var futures = IntStream.range(0, 10000)
                .mapToObj(i -> executor.submit(() -> handleRequest(i)))
                .toList();
            
            // 等待所有完成
            for (var future : futures) {
                future.get();
            }
        }
    }
    
    static String handleRequest(int id) {
        // 模拟I/O操作
        try {
            Thread.sleep(100);  // 虚拟线程休眠不阻塞底层OS线程
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
        return "Request " + id + " handled";
    }
}
```

**结构化并发（StructuredTaskScope）**：

```java
// 结构化并发API预览（JDK 26）
public record Order(String id, double amount) {}
public record User(String name, Order order) {}

// 结构化并发确保：要么所有子任务完成，要么全部取消
// 避免了"线程泄露"和"幽灵任务"问题
try (var scope = new StructuredTaskScope.ShutdownOnFailure()) {
    Future<User> user = scope.fork(() -> fetchUser(userId));
    Future<Order> order = scope.fork(() -> fetchLatestOrder(userId));
    
    scope.join();            // 等待所有子任务完成
    scope.throwIfFailed();   // 有失败则抛出
    
    return new Response(user.resultNow(), order.resultNow());
}
```

**▌ 迁移指南**

从传统线程模型迁移到虚拟线程的注意事项：

1. **ThreadLocal**：虚拟线程中ThreadLocal的基数大幅增加（可能创建数百万个），建议减少使用或改用ScopedValue
2. **同步块**：虚拟线程中使用`synchronized`会固定（pinning）底层OS线程，建议改用`ReentrantLock`
3. **线程池**：不要向虚拟线程传递传统线程池，使用`Executors.newVirtualThreadPerTaskExecutor()`
4. **监控链路**：83%的团队在ThreadLocal和监控链路上翻车，迁移前务必做好监控适配
