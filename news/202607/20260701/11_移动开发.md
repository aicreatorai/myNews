```swift
// Swift 6.3 并发示例（Actor成为一等公民）
actor BankAccount {
    private var balance: Double = 0
    
    func deposit(_ amount: Double) {
        balance += amount
    }
    
    func withdraw(_ amount: Double) async throws -> Double {
        guard balance >= amount else {
            throw BankError.insufficientFunds
        }
        balance -= amount
        return amount
    }
}

// 结构化并发简化
func transferFunds(from: BankAccount, to: BankAccount, amount: Double) async {
    async let withdrawal = try? from.withdraw(amount)
    async let deposit = to.deposit(amount)
    
    // 并发执行，自动管理
    _ = await (withdrawal, deposit)
}
```

```kotlin
fun requestDataConsent(context: Context) {

        ConsentPurpose.ANALYTICS,
        ConsentPurpose.PERSONALIZATION, 
        ConsentPurpose.ADVERTISING,
        ConsentPurpose.LOCATION_TRACKING
```
