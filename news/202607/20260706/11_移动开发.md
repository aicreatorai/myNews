class FoldableActivity : ComponentActivity() {
    private lateinit var windowManager: WindowManager
    private lateinit var foldFeature: FoldingFeature
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        windowManager = getSystemService(WindowManager::class.java)
        
        // 获取折叠特性
        val foldingFeatures = windowManager.currentWindowMetrics
            .windowInsets.getInsets(WindowInsets.Type.displayCutout())
            .displayFeatures
            .filterIsInstance<FoldingFeature>()
        
        foldingFeatures.firstOrNull()?.let { feature ->
            foldFeature = feature
            
            // 根据折叠状态调整布局
            when (feature.state) {
                FoldingFeature.State.FLAT -> {
                    // 完全展开
                    setContentView(R.layout.activity_tablet)
                }
                FoldingFeature.State.HALF_OPENED -> {
                    // 半开（帐篷模式）
                    setContentView(R.layout.activity_tent)
                }
                FoldingFeature.State.FOLDED -> {
                    // 折叠
                    setContentView(R.layout.activity_phone)
                }
            }
            
            // 监听折叠状态变化
            feature.registerCallback(object : FoldingFeature.Callback() {
                override fun onStateChanged(state: FoldingFeature.State) {
                    // 动态调整UI
                    updateLayoutForState(state)
                }
            })
        }
    }
}

    slice_strategy="expert_based"  # 按专家分片
)

# 保存分片模型
for i, sliced_model in enumerate(sliced_models):
    onnx.save(sliced_model, f"mixtral_slice_{i}.onnx")

# 移动端加载（Kotlin）
class MoEModelManager {
    private val ortEnv: OrtEnvironment
    private val sessionMap: MutableMap<String, OrtSession>
    
    init {
        ortEnv = OrtEnvironment.getEnvironment()
        sessionMap = mutableMapOf()
        
        // 按需加载专家分片
        val experts = listOf("expert_0", "expert_1", "expert_2", "expert_3")
        experts.forEach { expertName ->
            val modelPath = "models/mixtral_$expertName.onnx"
            val session = ortEnv.createSession(modelPath, OrtSession.SessionOptions())
            sessionMap[expertName] = session
        }
    }
    
    // 动态路由到不同专家
    fun predict(input: FloatArray, router: Router): FloatArray {
        val expertName = router.selectExpert(input)
        val session = sessionMap[expertName] ?: throw IllegalStateException()
        
        val inputTensor = OnnxTensor.createTensor(ortEnv, input)
        val results = session.run(mapOf("input" to inputTensor))
        
        return results["output"]?.value as FloatArray
    }
}
